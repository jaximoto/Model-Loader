cmake_minimum_required(VERSION 3.10.0)
project(waves VERSION 0.1.0 LANGUAGES C CXX)

set(ASSIMP_DIR "/home/jaxon/assimp")
find_library(ASSIMP, assimp HINTS "${ASSIMP_DIR}/bin" REQUIRED)
message(STATUS "Assimp found at: ${ASSIMP}")

add_executable(waves 
src/main.cpp
src/glad.h
src/glad.c
src/imgui.cpp
src/imgui_draw.cpp
src/imgui_tables.cpp
src/imgui_widgets.cpp
src/imgui_impl_glfw.cpp
src/imgui_impl_opengl3.cpp
src/imgui_demo.cpp
)


# link dirs for all libraries
link_directories(PUBLIC ${ASSIMP_DIR}/include ${ASSIMP_DIR}/bin)
link_libraries(PUBLIC ${ASSIMP})
include_directories(${ASSIMP_DIR}/include)
target_include_directories(waves PRIVATE include src PUBLIC ${ASSIMP_DIR}/include )

target_link_libraries(waves PUBLIC glfw ${ASSIMP})# Add include directories
